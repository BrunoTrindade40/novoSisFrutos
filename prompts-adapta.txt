src/mappers/caixaPaletizada.ts:
```typescript
import { CaixaPaletizada } from "../controllers/protocols";
import { CaixasPaletizadaMobileResult } from "../models/caixas";

// Função auxiliar para normalizar chaves do objeto para minúsculas
// Ex: Converte { "PalCai_Codigo": "123" } para { "palcai_codigo": "123" }
const normalizeKeys = (obj: any): any => {
  if (!obj) return {};
  const newObj: any = {};
  Object.keys(obj).forEach((key) => {
    newObj[key.toLowerCase()] = obj[key];
  });
  return newObj;
};

// Função auxiliar para garantir string segura
const safeString = (value: any): string => {
  if (value === null || value === undefined) return "";
  return String(value).trim();
};

export function toCaixaPaletizada(
  dbResultRaw: CaixasPaletizadaMobileResult
): CaixaPaletizada {
  if (!dbResultRaw) {
    throw new Error("Dados do produto inválidos ou vazios no Mapper.");
  }

  // 1. SOLUÇÃO SÊNIOR: Normaliza o objeto "cru" do banco
  // Isso garante que dbResult.palcai_codigo funcione mesmo se o banco mandar PalCai_Codigo
  const dbResult = normalizeKeys(dbResultRaw);

  return {
    codigoCaixa: safeString(dbResult.palcai_codigo),
    nomeProduto: safeString(dbResult.pro_descricao),
    dataColheita: safeString(dbResult.rom_dtcolheita),
    dataChegada: safeString(dbResult.rom_dtchegada || dbResult.rom_dtcolheita),
    produtorEmpresa: safeString(dbResult.emp_razaosocial), // Note: razaosocial tudo minúsculo após normalizar
    embalagem: safeString(dbResult.cai_descricao),
    nomeEmbalador: safeString(dbResult.cademb_nome),
    numeroRomaneio: safeString(dbResult.rom_romaneio),

    // Fallback robusto para talhão
    talhaoRomaneio: safeString(dbResult.rom_talhao),

    endereco: safeString(dbResult.endereco),
    cidade: safeString(dbResult.cidade),
    tamanhoProduto: safeString(dbResult.tam_descricao),
  };
}
```

```
SQL

USE [sisfrutos_BS]
GO
/****** Object:  StoredProcedure [dbo].[sp_p_CaixasPaletizada_mobile]    Script Date: 04/12/2025 13:23:45 ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
ALTER proc [dbo].[sp_p_CaixasPaletizada_mobile]
( @codcaixa varchar(15) , 
  @retornaDados int = 0,
  @resultado int =0 output)
as
declare @fkpalet int 
-- retorna o nr. do palet ou 0 se a caixa
-- não estiver paletizada
set @fkpalet=0

select @Fkpalet= fkpalet from paletcaixas with (readuncommitted)
         where PalCai_Codigo=@codcaixa

if @retornadados=1
   select fkpalet,
          idromaneio,
		  palcai_codigo,
          rom_romaneio,
		  rom_talhao,
		  rom_lote,
		  rom_dtcolheita,
	      emp_razaoSocial,
		  fkempresa,
		  emp_rua as endereco,
		  cid_Nome+'-'+emp_uf as cidade,
		  pro_descricao,
		  tam_descricao,
		  cai_descricao
		  Cademb_nome,
		  palcai_qtd,
		  palcai_peso
      from PaletCaixas with (readuncommitted)
	      left join sisfrutos..caixa on fkcaixa=idcaixa
	      left join sisfrutos..MarcaCaixa on Fkmarcacaixa = idMarcaCaixa
		  left join sisfrutos..tamanho on fktamanho = idtamanho
		  left join CadEmbalador on Fkembalador = IdCadembalador,
	    Romaneio with (readuncommitted)
	     left join sisfrutos..empresa with (readuncommitted) left join sisfrutos..cidade on FkCidade=idCidade
					on FkEmpresa = Idempresa 
         left join sisfrutos..produto with (readuncommitted) on fkproduto = idProduto
		 where fkromaneio=idromaneio
		   and palcai_codigo = @codcaixa
                     
set @resultado = isnull(@fkpalet,0)
select @resultado
return isnull(@resultado,0)
```
